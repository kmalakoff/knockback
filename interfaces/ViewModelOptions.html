    Extend the view model after observables are created.

    Example

    interface PersonViewModel {
      name: ko.Observable<string>;
      onSave: () => void;
    }
    const vm = viewModel<PersonViewModel>(model, {
      extend(vm, model) {
        vm.onSave = () => model?.save();
      },
    });
